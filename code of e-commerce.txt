// ─── STATE ────────────────────────────────────────────────────────────────
let cart = [];
let activeCategory = 'all';
let searchQuery = '';

// ─── INIT ────────────────────────────────────────────────────────────────
document.addEventListener('DOMContentLoaded', () => {
  renderCategories();
  renderProducts();
  document.getElementById('searchInput').addEventListener('keyup', e => {
    searchQuery = e.target.value.toLowerCase();
    renderProducts();
  });
});

// ─── CATEGORIES ───────────────────────────────────────────────────────────
function renderCategories() {
  const el = document.getElementById('categories');
  el.innerHTML = categories.map(c => `
    <button class="cat-btn ${c.id === activeCategory ? 'active' : ''}" onclick="filterCategory('${c.id}')">
      <span class="cat-emoji">${c.emoji}</span> ${c.label}
    </button>
  `).join('');
}

function filterCategory(id) {
  activeCategory = id;
  searchQuery = '';
  document.getElementById('searchInput').value = '';
  const cat = categories.find(c => c.id === id);
  document.getElementById('productsTitle').textContent = cat.emoji + ' ' + cat.label;
  renderCategories();
  renderProducts();
}

// ─── PRODUCTS ─────────────────────────────────────────────────────────────
function renderProducts() {
  const grid = document.getElementById('productsGrid');
  let filtered = activeCategory === 'all' ? products : products.filter(p => p.category === activeCategory);
  if (searchQuery) filtered = filtered.filter(p => p.name.toLowerCase().includes(searchQuery));

  if (!filtered.length) {
    grid.innerHTML = `<div style="grid-column:1/-1;text-align:center;padding:60px;color:var(--gray)"><div style="font-size:3rem">🔍</div><p style="margin-top:12px;font-weight:700">No products found</p></div>`;
    return;
  }
  grid.innerHTML = filtered.map((p, i) => `
    <div class="product-card" style="animation-delay:${i * 0.05}s">
      <div class="product-img">${p.emoji}</div>
      <div class="product-body">
        ${p.organic ? '<span class="badge-organic">🌿 Organic</span>' : ''}
        <div class="product-category">${p.category}</div>
        <div class="product-name">${p.name}</div>
        <div class="product-weight">📦 ${p.weight}</div>
        <div class="product-footer">
          <div class="product-price">${p.price.toLocaleString()} <span>RWF</span></div>
          <button class="btn-add" onclick="addToCart(${p.id})" title="Add to cart">
            <i class="fas fa-plus"></i>
          </button>
        </div>
      </div>
    </div>
  `).join('');
}

function searchProducts() {
  searchQuery = document.getElementById('searchInput').value.toLowerCase();
  renderProducts();
}

// ─── CART ─────────────────────────────────────────────────────────────────
function addToCart(id) {
  const product = products.find(p => p.id === id);
  const existing = cart.find(c => c.id === id);
  if (existing) existing.qty++;
  else cart.push({ ...product, qty: 1 });
  updateCartUI();
  showToast(`✅ ${product.name} added to cart`);
}

function removeFromCart(id) {
  cart = cart.filter(c => c.id !== id);
  updateCartUI();
}

function changeQty(id, delta) {
  const item = cart.find(c => c.id === id);
  if (!item) return;
  item.qty += delta;
  if (item.qty <= 0) removeFromCart(id);
  else updateCartUI();
}

function updateCartUI() {
  const count = cart.reduce((a, c) => a + c.qty, 0);
  document.getElementById('cartCount').textContent = count;
  renderCartItems();
}

function renderCartItems() {
  const el = document.getElementById('cartItems');
  const footer = document.getElementById('cartFooter');
  if (!cart.length) {
    el.innerHTML = `<div class="cart-empty"><i class="fas fa-shopping-basket"></i><p>Your cart is empty</p><p style="font-size:.82rem;margin-top:6px">Add some fresh groceries!</p></div>`;
    footer.style.display = 'none';
    return;
  }
  el.innerHTML = cart.map(item => `
    <div class="cart-item">
      <div class="cart-item-emoji">${item.emoji}</div>
      <div class="cart-item-info">
        <div class="cart-item-name">${item.name}</div>
        <div class="cart-item-price">${(item.price * item.qty).toLocaleString()} RWF</div>
      </div>
      <div class="qty-controls">
        <button class="qty-btn" onclick="changeQty(${item.id}, -1)">−</button>
        <span class="qty-num">${item.qty}</span>
        <button class="qty-btn" onclick="changeQty(${item.id}, 1)">+</button>
      </div>
      <button class="cart-remove" onclick="removeFromCart(${item.id})"><i class="fas fa-trash"></i></button>
    </div>
  `).join('');
  const subtotal = cart.reduce((a, c) => a + c.price * c.qty, 0);
  const total = subtotal + 500;
  document.getElementById('cartSubtotal').textContent = subtotal.toLocaleString() + ' RWF';
  document.getElementById('cartTotal').textContent = total.toLocaleString() + ' RWF';
  footer.style.display = 'block';
}

function toggleCart() {
  const sidebar = document.getElementById('cartSidebar');
  const overlay = document.getElementById('cartOverlay');
  sidebar.classList.toggle('open');
  overlay.classList.toggle('open');
  if (sidebar.classList.contains('open')) renderCartItems();
}

function resetCart() {
  cart = [];
  updateCartUI();
}

// ─── MOMO MODAL ───────────────────────────────────────────────────────────
function openMomoModal() {
  if (!cart.length) { showToast('⚠️ Your cart is empty!'); return; }
  const subtotal = cart.reduce((a, c) => a + c.price * c.qty, 0);
  const total = subtotal + 500;
  document.getElementById('payAmount').textContent = total.toLocaleString();
  const mini = document.getElementById('orderSummaryMini');
  mini.innerHTML = `
    <h4>📋 Order Summary</h4>
    ${cart.map(i => `<div class="summary-item"><span>${i.emoji} ${i.name} ×${i.qty}</span><span>${(i.price * i.qty).toLocaleString()} RWF</span></div>`).join('')}
    <div class="summary-item"><span>🚚 Delivery</span><span>500 RWF</span></div>
    <div class="summary-total"><span>Total</span><span>${total.toLocaleString()} RWF</span></div>
  `;
  showView('paymentFormView');
  document.getElementById('momoModal').classList.add('open');
  // Close cart
  document.getElementById('cartSidebar').classList.remove('open');
  document.getElementById('cartOverlay').classList.remove('open');
}

function closeModal() {
  document.getElementById('momoModal').classList.remove('open');
}

function showView(id) {
  ['paymentFormView','processingView','successView'].forEach(v => {
    document.getElementById(v).style.display = v === id ? 'block' : 'none';
  });
}

function setStep(n) {
  [1,2,3].forEach(i => {
    const el = document.getElementById('step'+i);
    el.className = 'step' + (i < n ? ' done' : i === n ? ' active' : '');
  });
}

function processPayment() {
  const name = document.getElementById('momoName').value.trim();
  const address = document.getElementById('momoAddress').value.trim();
  const phone = document.getElementById('momoPhone').value.trim();
  if (!name) { alert('Please enter your full name.'); return; }
  if (!address) { alert('Please enter your delivery address.'); return; }
  if (!phone || phone.length < 9) { alert('Please enter a valid MTN MoMo number (9 digits).'); return; }

  setStep(2);
  showView('processingView');

  // Simulate payment processing
  setTimeout(() => {
    setStep(3);
    showView('successView');
    const orderId = 'FM-' + Date.now().toString(36).toUpperCase() + '-KIC';
    document.getElementById('orderId').innerHTML = `📦 Order ID: <strong>${orderId}</strong>`;
  }, 3500);
}

// ─── TOAST ────────────────────────────────────────────────────────────────
function showToast(msg) {
  const t = document.getElementById('toast');
  document.getElementById('toastMsg').textContent = msg;
  t.classList.add('show');
  setTimeout(() => t.classList.remove('show'), 2800);
}
